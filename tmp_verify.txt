#!/usr/bin/env python3
"""DEVINE C10 rollback harness --- 4-tier atomic rollback tests on backup copies.

Each tier copies the backup DB to a temp file, applies a fake destructive
change, rolls it back atomically (BEGIN/COMMIT/except->rollback), then asserts
the original state is restored. All evidence rows are written into the TEMP DB
*only* --- never touches the live DB.
"""
from __future__ import annotations

import argparse
import json
import shutil
import sqlite3
import sys
import tempfile
from contextlib import contextmanager
from datetime import datetime, timezone
from pathlib import Path
from typing import Any, Iterator

import clm_qwen_schema_migration as schema_migration

HOME = Path(__file__).resolve().parents[1]
DEFAULT_RUN_ID = "devine_c10_rollback_v0"

PASS = "PASS"
FAIL = "FAIL"


def utc_now() -> str:
    return datetime.now(timezone.utc).isoformat()


def quote_ident(name: str) -> str:
    return '"' + name.replace('"', '""') + '"'


def columns(con: sqlite3.Connection, table: str) -> set[str]:
    return {str(r[1]) for r in con.execute(f"PRAGMA table_info({quote_ident(table)})")}


def table_count(con: sqlite3.Connection, table: str) -> int:
    return int(con.execute(f"SELECT COUNT(*) FROM {quote_ident(table)}").fetchone()[0])


def qwen_schema_applied(con: sqlite3.Connection) -> bool:
    cols = columns(con, "distilled_points")
    return "qwen_verify_status" in cols and "vector_eligible" in cols


def ensure_qwen_schema(con: sqlite3.Connection) -> None:
    if not qwen_schema_applied(con):
        schema_migration.apply_migration(con)


def write_evidence(
    con: sqlite3.Connection,
    run_id: str,
    check_name: str,
    status: str,
    detail: dict[str, Any],
) -> None:
    con.execute(
        """
        INSERT INTO harness_evidence(
            run_id, phase, check_name, status, severity, evidence_json, created_at, source, migration_batch_id
        )
        VALUES(?, 'RELEASE', ?, ?, 'INFO', ?, ?, 'clm_devine_rollback_harness', ?)
        """,
        (
            run_id,
            check_name,
            status,
            json.dumps(detail, sort_keys=True),
            utc_now(),
            run_id,
        ),
    )
    con.commit()


@contextmanager
def temp_copy(backup_db: Path) -> Iterator[Path]:
    with tempfile.NamedTemporaryFile(suffix=".db", delete=False) as tmp:
        tmp_path = Path(tmp.name)
    shutil.copy2(backup_db, tmp_path)
    try:
        yield tmp_path
    finally:
        if tmp_path.exists():
            tmp_path.unlink()


@contextmanager
def connect_scope(path: Path) -> Iterator[sqlite3.Connection]:
    con = sqlite3.connect(path)
    con.execute("PRAGMA journal_mode=WAL")
    con.row_factory = sqlite3.Row
    try:
        yield con
        con.commit()
    except Exception:
        con.rollback()
        raise
    finally:
        con.close()


# --- Tier 1 ---

def rollback_schema(con: sqlite3.Connection) -> dict[str, Any]:
    result = schema_migration.rollback(con)
    after_cols = columns(con, "distilled_points")
    qwen_cols = {"qwen_verify_status", "vector_eligible", "verify_source", "qwen_score", "qwen_scored_at", "migration_batch_id"}
    remnants = qwen_cols & after_cols
    score_table = schema_migration.table_exists(con, "qwen_semantic_scores")
    return {
        "ok": len(remnants) == 0 and not score_table,
        "remnant_columns": sorted(remnants),
        "qwen_semantic_scores_exists": score_table,
        "result": result,
    }


# --- Tier 2 ---

def rollback_verify_batch(con: sqlite3.Connection, run_id: str) -> dict[str, Any]:
    before_scores = table_count(con, "qwen_semantic_scores")
    before_points = [
        dict(r)
        for r in con.execute(
            "SELECT id, qwen_verify_status, vector_eligible, verify_source, qwen_score, qwen_scored_at, migration_batch_id "
            "FROM distilled_points WHERE migration_batch_id=? ORDER BY id",
            (run_id,),
        )
    ]
    ids = [p["id"] for p in before_points]
    try:
        con.execute("DELETE FROM qwen_semantic_scores WHERE migration_batch_id=?", (run_id,))
        con.execute(
            """
            UPDATE distilled_points
            SET qwen_verify_status = NULL,
                vector_eligible = 0,
                verify_source = 'legacy_heuristic',
                qwen_score = NULL,
                qwen_scored_at = NULL,
                migration_batch_id = NULL
            WHERE migration_batch_id = ?
            """,
            (run_id,),
        )
        con.commit()
    except Exception:
        con.rollback()
        raise
    after_scores = table_count(con, "qwen_semantic_scores")
    if ids:
        placeholders = ",".join("?" for _ in ids)
        reverted = list(
            con.execute(
                f"SELECT id, qwen_verify_status, vector_eligible, verify_source, qwen_score, qwen_scored_at "
                f"FROM distilled_points WHERE id IN ({placeholders}) ORDER BY id",
                ids,
            )
        )
    else:
        reverted = []
    all_null = all(
        r["qwen_verify_status"] is None
        and r["vector_eligible"] == 0
        and r["verify_source"] == "legacy_heuristic"
        and r["qwen_score"] is None
        and r["qwen_scored_at"] is None
        for r in reverted
    )
    return {
        "ok": (after_scores == before_scores - len(before_points)) and all_null,
        "scores_deleted": before_scores - after_scores,
        "points_reverted": len(reverted),
        "all_fields_null": all_null,
    }


# --- Tier 3 ---

def rollback_vector_reindex(con: sqlite3.Connection, model: str) -> dict[str, Any]:
    before_v2 = table_count(con, "vector_chunks_v2") if schema_migration.table_exists(con, "vector_chunks_v2") else 0
    before_legacy = table_count(con, "vector_chunks") if schema_migration.table_exists(con, "vector_chunks") else 0
    try:
        deleted = int(con.execute("DELETE FROM vector_chunks_v2 WHERE model=?", (model,)).rowcount)
        con.commit()
    except Exception:
        con.rollback()
        raise
    after_v2 = table_count(con, "vector_chunks_v2") if schema_migration.table_exists(con, "vector_chunks_v2") else 0
    after_legacy = table_count(con, "vector_chunks") if schema_migration.table_exists(con, "vector_chunks") else 0
    return {
        "ok": after_v2 == before_v2 - deleted and after_legacy == before_legacy,
        "deleted": deleted,
        "before_v2": before_v2,
        "after_v2": after_v2,
        "legacy_untouched": after_legacy == before_legacy,
    }


# --- Tier 4 ---

def row_ids(con: sqlite3.Connection, table: str, pk: str = "id") -> set[int]:
    return {int(r[0]) for r in con.execute(f"SELECT {quote_ident(pk)} FROM {quote_ident(table)}")}


def rollback_purge(
    con: sqlite3.Connection,
    backup_con: sqlite3.Connection,
    *,
    tables: tuple[str, ...] = ("distilled_points", "rag_chunks"),
) -> dict[str, Any]:
    results: dict[str, Any] = {"tables": {}, "total_restored": 0}
    for table in tables:
        current_ids = row_ids(con, table)
        backup_ids = row_ids(backup_con, table)
        purged_ids = sorted(backup_ids - current_ids)
        if not purged_ids:
            results["tables"][table] = {"restored": 0, "purged_ids": []}
            continue
        backup_rows = []
        for rid in purged_ids:
            row = dict(backup_con.execute(f"SELECT * FROM {quote_ident(table)} WHERE id=?", (rid,)).fetchone())
            backup_rows.append(row)
        try:
            for row in backup_rows:
                cols_list = list(row.keys())
                placeholders = ",".join("?" for _ in cols_list)
                quoted = ",".join(quote_ident(c) for c in cols_list)
                con.execute(f"INSERT OR IGNORE INTO {quote_ident(table)}({quoted}) VALUES({placeholders})", [row[c] for c in cols_list])
            con.commit()
        except Exception:
            con.rollback()
            raise
        after_ids = row_ids(con, table)
        results["tables"][table] = {"restored": len(purged_ids), "purged_ids": purged_ids, "after_count": len(after_ids)}
        results["total_restored"] += len(purged_ids)
    results["ok"] = results["total_restored"] >= 0
    return results


# --- Parity ---

ROW_COUNT_TABLES = (
    "schema_versions", "file_state", "distilled_points", "rag_chunks", "distill_runs",
    "vector_chunks", "memory_tags", "rag_chunks_fts", "rule_bridge_index",
    "filter_events", "memory_quality_scores", "memory_quality_digests",
    "guardrail_candidates", "raw_turns", "review_queue", "rerank_cache",
    "zone_index", "vector_chunks_backup_v1", "qwen_semantic_scores",
    "harness_evidence", "status_transition_log", "vector_chunks_v2",
)


def row_count_parity(con: sqlite3.Connection, manifest: dict[str, Any]) -> dict[str, Any]:
    expected = manifest.get("row_counts_backup", {})
    mismatches = {}
    for table in ROW_COUNT_TABLES:
        if not schema_migration.table_exists(con, table):
            mismatches[table] = {"expected": expected.get(table), "actual": None, "note": "table_missing"}
            continue
        actual = table_count(con, table)
        exp = expected.get(table)
        if exp is not None and actual != exp:
            mismatches[table] = {"expected": exp, "actual": actual}
    return {"ok": len(mismatches) == 0, "mismatches": mismatches, "mismatch_count": len(mismatches)}


# --- run_all ---

def run_all(
    backup_db: Path,
    manifest: dict[str, Any],
    *,
    run_id: str = DEFAULT_RUN_ID,
    verify_batch_run_id: str = "predevine_v0",
    vector_model: str = "local-hash-ngram-v2-qwen-gated",
) -> dict[str, Any]:
    results: dict[str, Any] = {}
    tiers_ok = 0

    with temp_copy(backup_db) as tmp:
        with connect_scope(tmp) as con:
            ensure_qwen_schema(con)
            tier1 = rollback_schema(con)
        results["rollback_schema"] = tier1
        status = PASS if tier1["ok"] else FAIL
        if tier1["ok"]: tiers_ok += 1
        with connect_scope(tmp) as ev_con:
            write_evidence(ev_con, run_id, "rollback_schema_test", status, tier1)

    with temp_copy(backup_db) as tmp:
        with connect_scope(tmp) as con:
            ensure_qwen_schema(con)
            existing = con.execute("SELECT COUNT(*) FROM qwen_semantic_scores WHERE migration_batch_id=?", (verify_batch_run_id,)).fetchone()[0]
            if existing == 0:
                con.execute("INSERT OR IGNORE INTO qwen_semantic_scores(distilled_point_id, score, verdict, migration_batch_id) SELECT id, 0.5, 'semantic_verified', ? FROM distilled_points LIMIT 3", (verify_batch_run_id,))
                con.execute("UPDATE distilled_points SET qwen_verify_status='semantic_verified', vector_eligible=1, verify_source='qwen_auto', qwen_score=0.5, qwen_scored_at=?, migration_batch_id=? WHERE id IN (SELECT id FROM distilled_points ORDER BY id LIMIT 3)", (utc_now(), verify_batch_run_id))
            tier2 = rollback_verify_batch(con, verify_batch_run_id)
        results["rollback_verify_batch"] = tier2
        status = PASS if tier2["ok"] else FAIL
        if tier2["ok"]: tiers_ok += 1
        with connect_scope(tmp) as ev_con:
            write_evidence(ev_con, run_id, "rollback_verify_batch_test", status, tier2)

    with temp_copy(backup_db) as tmp:
        with connect_scope(tmp) as con:
            ensure_qwen_schema(con)
            existing_v2 = table_count(con, "vector_chunks_v2") if schema_migration.table_exists(con, "vector_chunks_v2") else 0
            if existing_v2 == 0 and schema_migration.table_exists(con, "vector_chunks") and table_count(con, "vector_chunks") > 0:
                row = con.execute("SELECT * FROM vector_chunks LIMIT 1").fetchone()
                con.execute("INSERT INTO vector_chunks_v2(chunk_id, point_id, raw_turn_id, model, dim, vector_blob, text_sha256, mood, priority, created_at) VALUES(?, ?, ?, ?, ?, ?, ?, 'neutral', 9, ?)", (row["chunk_id"], row["point_id"], row["raw_turn_id"], vector_model, row["dim"], row["vector_blob"], row["text_sha256"], utc_now()))
            tier3 = rollback_vector_reindex(con, vector_model)
        results["rollback_vector_reindex"] = tier3
        status = PASS if tier3["ok"] else FAIL
        if tier3["ok"]: tiers_ok += 1
        with connect_scope(tmp) as ev_con:
            write_evidence(ev_con, run_id, "rollback_vector_reindex_test", status, tier3)

    with temp_copy(backup_db) as tmp:
        with sqlite3.connect(backup_db) as backup_con:
            backup_con.row_factory = sqlite3.Row
            parity = None
            with connect_scope(tmp) as con:
                ensure_qwen_schema(con)
                if table_count(con, "distilled_points") > 0:
                    tgt = con.execute("SELECT MIN(id) AS mid FROM distilled_points").fetchone()["mid"]
                    con.execute("DELETE FROM distilled_points WHERE id=?", (tgt,))
                    con.execute("DELETE FROM rag_chunks WHERE point_id=?", (tgt,))
                tier4 = rollback_purge(con, backup_con)
                parity = row_count_parity(con, manifest)
            results["rollback_purge"] = tier4
            status = PASS if tier4["ok"] else FAIL
            if tier4["ok"]: tiers_ok += 1
            with connect_scope(tmp) as ev_con:
                write_evidence(ev_con, run_id, "rollback_purge_test", status, tier4)
            results["row_count_parity"] = parity
            if parity and parity["ok"]: tiers_ok += 1

    results["summary"] = {"tiers_ok": tiers_ok, "total_tiers": 5, "run_id": run_id}
    return results


# --- CLI ---

def parse_args(argv: list[str] | None = None) -> argparse.Namespace:
    parser = argparse.ArgumentParser(description="DEVINE C10 rollback harness")
    parser.add_argument("--backup-db", required=True, type=Path)
    parser.add_argument("--manifest", required=True, type=Path)
    parser.add_argument("--run-id", default=DEFAULT_RUN_ID)
    parser.add_argument("--verify-batch-run-id", default="predevine_v0")
    parser.add_argument("--vector-model", default="local-hash-ngram-v2-qwen-gated")
    parser.add_argument("--run-all", action="store_true")
    return parser.parse_args(argv)


def main(argv: list[str] | None = None) -> int:
    args = parse_args(argv)
    if args.run_all:
        manifest = json.loads(args.manifest.read_text(encoding="utf-8"))
        results = run_all(backup_db=args.backup_db, manifest=manifest, run_id=args.run_id, verify_batch_run_id=args.verify_batch_run_id, vector_model=args.vector_model)
        print(json.dumps(results, ensure_ascii=False, indent=2))
        return 0 if results.get("summary", {}).get("tiers_ok", 0) == 5 else 1
    return 1


if __name__ == "__main__":
    raise SystemExit(main())
